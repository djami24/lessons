rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }
    function isAdmin() {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }
    // Reads a nested course-payment status safely, defaulting to 'none'
    // when the student doc (or the coursePayments map) doesn't have it yet.
    function courseStatus(data, courseId) {
      return data.get('coursePayments', {}).get(courseId, 'none');
    }
    // A student may move any course's payment status to 'pending', or leave
    // it unchanged — but never jump straight to 'approved' themselves.
    function validCoursePaymentsUpdate() {
      return courseStatus(request.resource.data, 'ielts') == courseStatus(resource.data, 'ielts') ||
             courseStatus(request.resource.data, 'ielts') == 'pending';
    }
    function validCoursePaymentsUpdateCefr() {
      return courseStatus(request.resource.data, 'cefr') == courseStatus(resource.data, 'cefr') ||
             courseStatus(request.resource.data, 'cefr') == 'pending';
    }
    // Writing (insho) — separate paid add-on, same self-service-to-'pending'-only rule.
    function validCoursePaymentsUpdateWriting() {
      return courseStatus(request.resource.data, 'writing') == courseStatus(resource.data, 'writing') ||
             courseStatus(request.resource.data, 'writing') == 'pending';
    }
 
    // Admin accounts — document ID = the admin's Firebase Auth UID.
    // Create these manually in the Firestore console (see setup guide).
    match /admins/{adminId} {
      allow read: if isSignedIn() && request.auth.uid == adminId;
      allow write: if false; // only editable from the Firebase console
    }
 
    // Site settings (colors, brand name, logo, homepage text) — editable
    // by the admin from the dashboard. Publicly readable because the
    // marketing homepage (index.html) has no signed-in user.
    match /settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();

      // Homepage "rate the site" widget: any visitor (signed in or not)
      // may cast exactly one vote per request, worth 1-5 points, by
      // bumping ratingSum/ratingCount on settings/site — nothing else.
      // Covers the very first vote (doc doesn't exist yet)...
      allow create: if !isAdmin() && docId == 'site' &&
        request.resource.data.keys().hasOnly(['ratingSum', 'ratingCount']) &&
        request.resource.data.ratingCount == 1 &&
        request.resource.data.ratingSum >= 1 && request.resource.data.ratingSum <= 5;

      // ...and every vote after that (doc already exists).
      allow update: if !isAdmin() && docId == 'site' &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['ratingSum', 'ratingCount']) &&
        request.resource.data.ratingCount == resource.data.get('ratingCount', 0) + 1 &&
        request.resource.data.ratingSum >= resource.data.get('ratingSum', 0) + 1 &&
        request.resource.data.ratingSum <= resource.data.get('ratingSum', 0) + 5;
    }
 
    // Blog posts ("Djami's Blog") — admin writes them from the dashboard;
    // publicly readable since the homepage preview and blog.html detail
    // page both run with no signed-in user.
    match /blogs/{blogId} {
      allow read: if true;
      allow write: if isAdmin();

      // Public like / dislike buttons on blog.html: any visitor (signed in
      // or not) may bump exactly ONE of likeCount/dislikeCount by exactly 1
      // per request — nothing else on the post can change this way. The
      // front end also remembers the vote in localStorage so the same
      // browser doesn't vote twice, but that's a UX nicety, not security —
      // this rule is what actually enforces "+1 to one counter, nothing more".
      allow update: if !isAdmin() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['likeCount']) &&
        request.resource.data.likeCount == resource.data.get('likeCount', 0) + 1;

      allow update: if !isAdmin() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['dislikeCount']) &&
        request.resource.data.dislikeCount == resource.data.get('dislikeCount', 0) + 1;

      // Comments on a blog post (blog.html): open to any visitor, signed in
      // or not — no moderation queue, comments appear immediately. A
      // comment is just { name, text, createdAt } and can never be edited
      // after posting. Only an admin can delete one (spam/moderation),
      // from the admin dashboard's "Kommentlar" view.
      match /comments/{commentId} {
        allow read: if true;

        allow create: if request.resource.data.keys().hasOnly(['name', 'text', 'createdAt']) &&
          request.resource.data.name is string &&
          request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 60 &&
          request.resource.data.text is string &&
          request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 1000 &&
          request.resource.data.createdAt == request.time;

        allow update: if false;

        allow delete: if isAdmin();
      }
    }

    // General English: fixed level docs (beginner / elementary /
    // pre-intermediate / intermediate), each holding an embedded `lessons`
    // array (topic, homework, quiz questions) and an optional admin lock
    // (locked, lockedMessage). Any signed-in user may read — needed to show
    // locked previews to unpaid students and to fetch quiz questions.
    // Only admins may create/edit/delete lessons or toggle the lock.
    match /levels/{levelId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
 
    // IELTS / CEFR: fixed course docs (ielts / cefr), each holding an
    // embedded `sections` map (listening / reading / speaking), and each
    // section holding its own `topics` array plus an optional admin lock.
    // Same read/write shape as `levels`.
    match /courses/{courseId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    // Placement test ("Darajani bilish testi") — single doc placementTest/main
    // holding { questions:[{question,options,correctIndex}], levels:[{min,max,label}] }.
    // Admin manages questions and the score→level scale; any signed-in
    // student may read it to take the test. Same read/write shape as `levels`.
    match /placementTest/{docId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
 
    // Student accounts — document ID = the student's Firebase Auth UID.
    match /students/{studentId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
 
      // A signed-in user may create ONLY their own document, always unpaid
      // and with empty progress (course access + progress are never
      // self-assigned at creation time).
      allow create: if isSignedIn() && request.auth.uid == studentId &&
        request.resource.data.paymentStatus == 'none' &&
        request.resource.data.progress == {} &&
        request.resource.data.keys().hasOnly(['name','email','paymentStatus','progress','createdAt']);
 
      // Admin can change anything (approve/revoke payment for General
      // English, IELTS, or CEFR; edit; delete).
      // A student may only:
      //   - update their own quiz progress,
      //   - save their own placement-test result (placementResult — score,
      //     total, level, takenAt; no validation needed, it's just a
      //     self-reported summary the student and admin can both see), and
      //   - flip their OWN paymentStatus / coursePayments.ielts /
      //     coursePayments.cefr to 'pending' (e.g. after clicking "To'lov
      //     qildim") — they can NEVER set any of these to 'approved'
      //     themselves; only an admin can grant course access.
      allow update: if isAdmin() || (
        isSignedIn() && request.auth.uid == studentId &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['progress','paymentStatus','paymentRequestedAt','coursePayments','coursePaymentRequestedAt','placementResult']) &&
        request.resource.data.paymentStatus in ['pending', resource.data.paymentStatus] &&
        validCoursePaymentsUpdate() &&
        validCoursePaymentsUpdateCefr() &&
        validCoursePaymentsUpdateWriting()
      );
 
      allow delete: if isAdmin();
    }

    // Writing (insho) submissions — a student picks a question (under
    // courses/writing) and submits an essay; only an admin may grade it.
    // Document shape: { studentId, studentName, trackId, trackName,
    //   questionId, questionTitle, questionText, essayText, status:
    //   'pending'|'graded', score, comment, submittedAt, gradedAt }
    match /writingSubmissions/{submissionId} {
      // A student may read only their own submissions; admin reads all.
      allow read: if isAdmin() || (isSignedIn() && resource.data.studentId == request.auth.uid);

      // A student may submit an essay for THEMSELVES only, always starting
      // out ungraded (no score/comment yet) — and only if their Writing
      // course access has been approved by an admin.
      allow create: if isSignedIn() &&
        request.resource.data.studentId == request.auth.uid &&
        request.resource.data.status == 'pending' &&
        request.resource.data.score == null &&
        request.resource.data.comment == null &&
        get(/databases/$(database)/documents/students/$(request.auth.uid)).data
          .get('coursePayments', {}).get('writing', 'none') == 'approved';

      // Only an admin can grade (set status/score/comment/gradedAt).
      allow update: if isAdmin() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','score','comment','gradedAt']);

      // Admin can delete any submission; a student may withdraw their own
      // submission only while it's still ungraded.
      allow delete: if isAdmin() || (isSignedIn() && resource.data.studentId == request.auth.uid && resource.data.status == 'pending');
    }

    // Kundalik (homework diary) — admin creates groups and daily homework;
    // any authenticated user (student) may read.
    match /kundalik/{groupId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();

      match /vazifalari/{hwId} {
        allow read: if isSignedIn();
        allow write: if isAdmin();
      }
    }

  }
}
