rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }
    function isAdmin() {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }
    function courseStatus(data, courseId) {
      return data.get('coursePayments', {}).get(courseId, 'none');
    }
    function validCoursePaymentsUpdate() {
      return courseStatus(request.resource.data, 'ielts') == courseStatus(resource.data, 'ielts') ||
             courseStatus(request.resource.data, 'ielts') == 'pending';
    }
    function validCoursePaymentsUpdateCefr() {
      return courseStatus(request.resource.data, 'cefr') == courseStatus(resource.data, 'cefr') ||
             courseStatus(request.resource.data, 'cefr') == 'pending';
    }
    function validCoursePaymentsUpdateWriting() {
      return courseStatus(request.resource.data, 'writing') == courseStatus(resource.data, 'writing') ||
             courseStatus(request.resource.data, 'writing') == 'pending';
    }
    function validKundalikGroupUpdate() {
      return resource.data.get('kundalikGroup', null) == null ||
             request.resource.data.kundalikGroup == resource.data.kundalikGroup;
    }
    function isLinkedParentFor(childCode) {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/parents/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/parents/$(request.auth.uid)).data.linkedStudentId == childCode;
    }

    match /admins/{adminId} {
      allow read: if isSignedIn() && request.auth.uid == adminId;
      allow write: if false;
    }
    match /settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /blogs/{blogId} {
      allow read: if true;
      allow write: if isAdmin();
      match /comments/{commentId} {
        allow read: if true;
        allow create: if request.resource.data.keys().hasOnly(['name', 'text', 'createdAt']) &&
          request.resource.data.name is string &&
          request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 60 &&
          request.resource.data.text is string &&
          request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 1000 &&
          request.resource.data.createdAt == request.time;
        allow delete: if isAdmin();
      }
    }
    match /levels/{levelId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /courses/{courseId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /placementTest/{docId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /students/{studentId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
      allow create: if isSignedIn() && request.auth.uid == studentId;
      allow update: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
      allow delete: if isAdmin();
      match /savedMocks/{mockId} {
        allow read, write: if isSignedIn() && request.auth.uid == studentId;
      }
    }
    match /parents/{parentId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid == parentId);
      allow create: if isSignedIn() && request.auth.uid == parentId;
      allow update: if isAdmin() || (isSignedIn() && request.auth.uid == parentId);
      allow delete: if isAdmin();
    }
    match /writingSubmissions/{submissionId} {
      allow read: if isAdmin() || (isSignedIn() && resource.data.studentId == request.auth.uid);
      allow create: if isSignedIn() && request.resource.data.studentId == request.auth.uid;
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }
    match /kundalik/{groupId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
      match /vazifalari/{hwId} {
        allow read: if isSignedIn();
        allow write: if isAdmin();
      }
    }
    match /gradebook/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /stats/{docId} {
      allow read: if isAdmin();
      allow create, update: if true;
    }
    match /notifications/{notifId} {
      allow read: if true;
      allow create, delete: if isAdmin();
      allow update: if false;
    }
    match /progress/{studentId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
      allow write: if isAdmin();
      match /entries/{entryId} {
        allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
        allow write: if isAdmin();
      }
    }
    match /oquv_groups/{groupId} {
      allow read: if true;
      allow write: if isAdmin();
      match /students/{studentId} {
        allow read, write: if isAdmin();
      }
    }
    match /oquv_progress/{studentDocId} {
      allow read: if true;
      allow write: if isAdmin();
      match /entries/{entryId} {
        allow read: if true;
        allow write: if isAdmin();
      }
    }
    match /chats/{studentId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
      allow create, update: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
      allow delete: if isAdmin();
      match /messages/{msgId} {
        allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
        allow create: if isSignedIn();
        allow update: if isAdmin();
        allow delete: if isAdmin();
      }
    }
    match /presence/{uid} {
      allow read: if isSignedIn();
      allow write: if isSignedIn();
    }
    match /parentChats/{childCode} {
      allow read: if isAdmin() || isLinkedParentFor(childCode);
      allow create, update: if isAdmin() || isLinkedParentFor(childCode);
      allow delete: if isAdmin();
      match /messages/{msgId} {
        allow read: if isAdmin() || isLinkedParentFor(childCode);
        allow create: if isAdmin() || isLinkedParentFor(childCode);
        allow update: if isAdmin();
        allow delete: if isAdmin();
      }
    }
    match /activity_log/{logId} {
      allow create: if true;
      allow read, delete: if isAdmin();
      allow update: if false;
    }

    // ===== READING MOCKS — hamma o'qiy oladi =====
    match /reading_mocks/{mockId} {
      allow read: if true;
      allow write: if isAdmin();
      match /passages/{passageId} {
        allow read: if true;
        allow write: if isAdmin();
        match /groups/{groupId} {
          allow read: if true;
          allow write: if isAdmin();
          match /questions/{questionId} {
            allow read: if true;
            allow write: if isAdmin();
          }
        }
      }
    }

    match /reading_mock_attempts/{attemptId} {
      allow read: if isAdmin() || (isSignedIn() && resource.data.studentId == request.auth.uid);
      allow create: if isSignedIn() && request.resource.data.studentId == request.auth.uid;
      allow update, delete: if isAdmin();
    }
  }
}
