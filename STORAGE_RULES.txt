rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // Lesson video/material files, uploaded by admins from admin/dashboard.html.
    match /lesson-files/{folderId}/{fileName} {
      // Any signed-in user (student or admin) may view/download lesson files.
      allow read: if request.auth != null;

      // Only admins may upload/replace/delete files, and only within
      // the size limits enforced client-side (300MB video / 30MB material).
      allow write: if request.auth != null &&
        firestore.exists(/databases/(default)/documents/admins/$(request.auth.uid)) &&
        request.resource.size < 300 * 1024 * 1024;
    }

    // Everything else stays locked down by default.
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
