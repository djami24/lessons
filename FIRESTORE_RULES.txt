rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }
    function isAdmin() {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }
    function courseStatus(data, courseId) {
      return data.get('coursePayments', {}).get(courseId, 'none');
    }
    function validCoursePaymentsUpdate() {
      return courseStatus(request.resource.data, 'ielts') == courseStatus(resource.data, 'ielts') ||
             courseStatus(request.resource.data, 'ielts') == 'pending';
    }
    function validCoursePaymentsUpdateCefr() {
      return courseStatus(request.resource.data, 'cefr') == courseStatus(resource.data, 'cefr') ||
             courseStatus(request.resource.data, 'cefr') == 'pending';
    }
    function validCoursePaymentsUpdateWriting() {
      return courseStatus(request.resource.data, 'writing') == courseStatus(resource.data, 'writing') ||
             courseStatus(request.resource.data, 'writing') == 'pending';
    }
    function validKundalikGroupUpdate() {
      return resource.data.get('kundalikGroup', null) == null ||
             request.resource.data.kundalikGroup == resource.data.kundalikGroup;
    }

    function isLinkedParentFor(childCode) {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/parents/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/parents/$(request.auth.uid)).data.linkedStudentId == childCode;
    }

    match /admins/{adminId} {
      allow read: if isSignedIn() && request.auth.uid == adminId;
      allow write: if false;
    }

    match /settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();

      allow create: if !isAdmin() && docId == 'site' &&
        request.resource.data.keys().hasOnly(['ratingSum', 'ratingCount']) &&
        request.resource.data.ratingCount == 1 &&
        request.resource.data.ratingSum >= 1 && request.resource.data.ratingSum <= 5;

      allow update: if !isAdmin() && docId == 'site' &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['ratingSum', 'ratingCount']) &&
        request.resource.data.ratingCount == resource.data.get('ratingCount', 0) + 1 &&
        request.resource.data.ratingSum >= resource.data.get('ratingSum', 0) + 1 &&
        request.resource.data.ratingSum <= resource.data.get('ratingSum', 0) + 5;
    }

    match /blogs/{blogId} {
      allow read: if true;
      allow write: if isAdmin();

      allow update: if !isAdmin() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['likeCount']) &&
        request.resource.data.likeCount == resource.data.get('likeCount', 0) + 1;

      allow update: if !isAdmin() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['dislikeCount']) &&
        request.resource.data.dislikeCount == resource.data.get('dislikeCount', 0) + 1;

      match /comments/{commentId} {
        allow read: if true;

        allow create: if request.resource.data.keys().hasOnly(['name', 'text', 'createdAt']) &&
          request.resource.data.name is string &&
          request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 60 &&
          request.resource.data.text is string &&
          request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 1000 &&
          request.resource.data.createdAt == request.time;

        allow update: if !isAdmin() &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['likeCount']) &&
          request.resource.data.likeCount == resource.data.get('likeCount', 0) + 1;

        allow update: if !isAdmin() &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['dislikeCount']) &&
          request.resource.data.dislikeCount == resource.data.get('dislikeCount', 0) + 1;

        allow delete: if isAdmin();
      }
    }

    match /levels/{levelId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    match /courses/{courseId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    match /placementTest/{docId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    match /students/{studentId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);

      allow create: if isSignedIn() && request.auth.uid == studentId &&
        request.resource.data.paymentStatus == 'none' &&
        request.resource.data.progress == {} &&
        request.resource.data.keys().hasOnly(['name','email','phone','studentType','paymentStatus','progress','createdAt']);

      allow update: if isAdmin() || (
        isSignedIn() && request.auth.uid == studentId &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['progress','paymentStatus','paymentRequestedAt','coursePayments','coursePaymentRequestedAt','placementResult','kundalikGroup','gradebookName','oquvProgressCode']) &&
        request.resource.data.paymentStatus in ['pending', resource.data.paymentStatus] &&
        validCoursePaymentsUpdate() &&
        validCoursePaymentsUpdateCefr() &&
        validCoursePaymentsUpdateWriting() &&
        validKundalikGroupUpdate()
      );

      allow delete: if isAdmin();
    }

    match /parents/{parentId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid == parentId);
      allow create: if isSignedIn() && request.auth.uid == parentId;
      allow update: if isAdmin() || (isSignedIn() && request.auth.uid == parentId);
      allow delete: if isAdmin();
    }

    match /writingSubmissions/{submissionId} {
      allow read: if isAdmin() || (isSignedIn() && resource.data.studentId == request.auth.uid);

      allow create: if isSignedIn() &&
        request.resource.data.studentId == request.auth.uid &&
        request.resource.data.status == 'pending' &&
        request.resource.data.score == null &&
        request.resource.data.comment == null &&
        get(/databases/$(database)/documents/students/$(request.auth.uid)).data
          .get('coursePayments', {}).get('writing', 'none') == 'approved';

      allow update: if isAdmin() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','score','comment','gradedAt']);

      allow delete: if isAdmin() || (isSignedIn() && resource.data.studentId == request.auth.uid && resource.data.status == 'pending');
    }

    match /kundalik/{groupId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();

      match /vazifalari/{hwId} {
        allow read: if isSignedIn();
        allow write: if isAdmin();
      }
    }

    match /gradebook/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /stats/{docId} {
      allow read: if isAdmin();

      allow create: if docId == 'visits' &&
        request.resource.data.keys().hasOnly(['count', 'lastVisit']) &&
        request.resource.data.count == 1 &&
        request.resource.data.lastVisit == request.time;

      allow update: if docId == 'visits' &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['count', 'lastVisit']) &&
        request.resource.data.count == resource.data.get('count', 0) + 1 &&
        request.resource.data.lastVisit == request.time;
    }

    match /notifications/{notifId} {
      allow read: if isSignedIn() || true;
      allow create, delete: if isAdmin();
      allow update: if false;
    }

    match /progress/{studentId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
      allow write: if isAdmin();

      match /entries/{entryId} {
        allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
        allow write: if isAdmin();
      }
    }

    match /oquv_groups/{groupId} {
      allow read: if true;
      allow write: if isAdmin();

      match /students/{studentId} {
        allow read, write: if isAdmin();
      }
    }

    match /oquv_progress/{studentDocId} {
      allow read: if true;
      allow write: if isAdmin();

      match /entries/{entryId} {
        allow read: if true;
        allow write: if isAdmin();
      }
    }

    match /chats/{studentId} {
      allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
      allow create, update: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);
      allow delete: if isAdmin();

      match /messages/{msgId} {
        allow read: if isAdmin() || (isSignedIn() && request.auth.uid == studentId);

        allow create: if isSignedIn() && request.auth.uid == studentId &&
          request.resource.data.senderRole == 'student' &&
          request.resource.data.text is string &&
          request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 1000;

        allow create: if isAdmin() &&
          request.resource.data.senderRole == 'teacher';

        allow update: if isAdmin() || (
          isSignedIn() && request.auth.uid == studentId &&
          resource.data.senderRole == 'teacher' &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['read']) &&
          request.resource.data.read == true
        );

        allow delete: if isAdmin();
      }
    }

    // Presence — online / last-seen status for the student<->teacher chat.
    // presence/{uid}: a student writes only to presence/{their own uid};
    // any signed-in admin session writes to the single shared
    // presence/teacher-status doc (so "teacher online" reflects whichever
    // admin currently has the dashboard open). Everyone signed in can
    // read presence, since both sides need to see the other's status.
    match /presence/{uid} {
      allow read: if isSignedIn();
      allow write: if isSignedIn() && (
        request.auth.uid == uid ||
        (uid == 'teacher-status' && isAdmin())
      );
    }

    match /parentChats/{childCode} {
      allow read: if isAdmin() || isLinkedParentFor(childCode);
      allow create, update: if isAdmin() || isLinkedParentFor(childCode);
      allow delete: if isAdmin();

      match /messages/{msgId} {
        allow read: if isAdmin() || isLinkedParentFor(childCode);

        allow create: if isLinkedParentFor(childCode) &&
          request.resource.data.senderRole == 'parent' &&
          request.resource.data.text is string &&
          request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 1000;

        allow create: if isAdmin() &&
          request.resource.data.senderRole == 'teacher';

        allow update: if isAdmin() || (
          isLinkedParentFor(childCode) &&
          resource.data.senderRole == 'teacher' &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['read']) &&
          request.resource.data.read == true
        );

        allow delete: if isAdmin();
      }
    }

    // Faollik jurnali — talaba va ota-onalar kirish/ro'yxatdan o'tish tarixi,
    // shuningdek tizimga kirmagan mehmonlarning tashrifi (guest-visit-log.js).
    match /activity_log/{logId} {
      // Tizimga kirgan foydalanuvchi (talaba/ota-ona) — login/register yozuvi
      allow create: if isSignedIn() &&
        request.resource.data.keys().hasAll(['role', 'action', 'at']) &&
        request.resource.data.role in ['student', 'parent'] &&
        request.resource.data.at == request.time;

      // Tizimga kirmagan mehmon — sahifaga tashrif (guest-visit-log.js)
      allow create: if !isSignedIn() &&
        request.resource.data.role == 'guest' &&
        request.resource.data.action == 'visit' &&
        request.resource.data.at == request.time &&
        request.resource.data.keys().hasOnly(
          ['role','action','page','path','at','ua','browser','deviceId','isReturning','ip','city','country']
        ) &&
        request.resource.data.ua is string && request.resource.data.ua.size() <= 500;

      allow read, delete: if isAdmin();
      allow update: if false;
    }

  }
}
